# The game code is not in this repository: run the code generator first (tools/codegen.ps1), which writes generated/

# Windows PC:
#   cmake --preset win-pc && cmake --build --preset win-pc        (from app/, inside tools\win\env.ps1)
# Nintendo Switch (devkitA64), later:
#   cmake -S app -B out/nx -G Ninja -DCMAKE_BUILD_TYPE=Release \
#     -DCMAKE_TOOLCHAIN_FILE=tools/switch/cmake/switch-devkitA64.cmake -DREXGLUE_SWITCH_NVK_SDK=<mesa sdk>

# Console, 2026-10-13: GCC's AArch64 early register allocation pass (early_ra; -mearly-ra=all is the default from
# -O2 in devkitA64's GCC 16.1) miscompiled the game's streaming copies for an unaligned source (sub_8260E9C0,
# sub_8260E164: 72 two-register table lookups, 65 vectors live). The optimised GIMPLE was right; after the pass 38
# of the lookups were gone, the register with the byte-reversal constant was overwritten by a data load inside the
# loop, and every atlas copied through them was garbage from its second megabyte on. With the pass off the function
# is complete at -O2 and -O3 (docs/feedingfrenzy2-notes.md, "Console: large atlases"). Off for everything built for the
# console, the SDK included (directory scope, before the SDK is added), and at the link, where an LTO build
# generates its code. tools/check_switch_build.py checks that every compile command has it.
if(REXGLUE_PLATFORM_SWITCH AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14)
    add_compile_options(-mearly-ra=none)
    add_link_options(-mearly-ra=none)
endif()

# The SDK compiles itself with these (directory scope), but they do not reach the generated game code, which is
# built in this directory. GCC's default for GNU C++ is -ffp-contract=fast: it fuses separate PPC fmul/fadd into
# FMA and results drift from the Xbox 360. Strict aliasing is unsafe for recompiled code (guest memory is read and
# written at every width through the same addresses). Checked in build.ninja 2026-10-06: the Switch objects had both;
# the PC (clang) ones had neither, so Clang is included now (banjo-nx's check, relayed by the Mushihimesama session).
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target feedingfrenzy2 feedingfrenzy2_recomp)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE -ffp-contract=off -fno-strict-aliasing)
        endif()
    endforeach()
    # 2026-10-07 (masseffect-nx ships them; futari-nx: menus 426/427 and its fade 8/8 identical, verify 0): sqrt and
    # friends without the errno branch, FP operations that may be if-converted or vectorised. The guest's FPSCR here
    # is only the rounding and flush modes (sdk/include/rex/ppc/context.h): no FP exception flag is ever read.
    # OFF in Peggle 2 until a quiet-PC lockstep settles it: two captures with them missed the menu fade-out of game
    # frames 1022-1047 that the CPU-off baselines reproduce, but a capture WITHOUT them made under the same load (a
    # Switch LTO build running) missed it too, so the cause is not shown to be the flags. Verify mode: 0 differences.
    option(FEEDINGFRENZY2_NO_MATH_ERRNO "-fno-math-errno -fno-trapping-math on the game code and the app" OFF)
    if(FEEDINGFRENZY2_NO_MATH_ERRNO)
        foreach(target feedingfrenzy2 feedingfrenzy2_recomp)
            if(TARGET ${target})
                target_compile_options(${target} PRIVATE -fno-math-errno -fno-trapping-math)
            endif()
        endforeach()
    endif()
    # sa2-nx's second CPU round (2026-10-12), the game code only: `sqrt` without the errno check (a compare and a
    # branch to the library after every fsqrt). Values are unchanged: nothing reads errno. The same flag on the file
    # with the localized copies (GCC does not inline between two float functions that differ in -fmath-errno).
    # -fno-trapping-math stays out (sa2-nx's audit: it lets the compiler run a float operation earlier than the code
    # does, across a flush-mode switch it does not know the operation depends on).
    option(FEEDINGFRENZY2_GAME_NO_MATH_ERRNO "-fno-math-errno on the recompiled game code and the localized copies" ON)
    if(FEEDINGFRENZY2_GAME_NO_MATH_ERRNO AND NOT FEEDINGFRENZY2_NO_MATH_ERRNO)
        if(TARGET feedingfrenzy2_recomp)
            target_compile_options(feedingfrenzy2_recomp PRIVATE -fno-math-errno)
        endif()
        set_property(SOURCE ${FEEDINGFRENZY2_HOT_SOURCES} APPEND PROPERTY COMPILE_OPTIONS -fno-math-errno)
    endif()
endif()

# Switch compiler set (sa2-nx's round 3, UnleashedRecomp-NX's set without PGO; the user's choice for the ports):
#   - LTO: the recompiled game (feedingfrenzy2_recomp) and the app (renderer, natives) are compiled to GIMPLE
#     (-flto -fno-fat-lto-objects) and optimised as one program at the link (-flto=N -flto-partition=balanced, N
#     parallel LTRANS jobs, kept low for the build machine). Calls between unhooked game functions are direct
#     (tools/perf_codegen_post.py), so small game functions inline across the 122 generated files. SDK libraries stay out.
#   - -fipa-pta at compile and link (interprocedural points-to analysis).
#   - The link repeats the exactness flags (the LTRANS stage must not fuse or alias what the compile stage kept apart),
#     and xxHash is built with XXH_FORCE_MEMORY_ACCESS=0 (nfsmw-nx's LTO pitfall: its default reads violate strict
#     aliasing once inlined into a key builder).
#   - Line tables (-g1, DWARF 4) for the app's own code: console profiles map to source lines on the archived ELF; -g
#     does not change GCC's code generation.
if(REXGLUE_PLATFORM_SWITCH)
    # peggle1-nx (2026-10-21): off, releases included (the user: "We don't need LTO or PGO because the game is extremely
    # lightweight").
    option(FEEDINGFRENZY2_SWITCH_LTO "Link-time optimisation of the game code and the app (Switch)" OFF)
    set(FEEDINGFRENZY2_SWITCH_LTO_JOBS "2" CACHE STRING "Parallel LTRANS jobs of the Switch LTO link")
    option(FEEDINGFRENZY2_SWITCH_IPA_PTA "-fipa-pta at the Switch LTO link" ON)
    option(FEEDINGFRENZY2_SWITCH_LINE_INFO "Line tables (-g1) for the app's own code in the Switch ELF" ON)
    target_compile_definitions(feedingfrenzy2 PRIVATE XXH_FORCE_MEMORY_ACCESS=0)
    if(FEEDINGFRENZY2_SWITCH_LTO)
        set(_feedingfrenzy2_lto_compile -flto -fno-fat-lto-objects)
        set(_feedingfrenzy2_lto_link -flto=${FEEDINGFRENZY2_SWITCH_LTO_JOBS} -flto-partition=balanced -ffp-contract=off
            -fno-strict-aliasing)
        if(FEEDINGFRENZY2_NO_MATH_ERRNO)
            list(APPEND _feedingfrenzy2_lto_link -fno-math-errno -fno-trapping-math)
        endif()
        if(FEEDINGFRENZY2_SWITCH_IPA_PTA)
            list(APPEND _feedingfrenzy2_lto_compile -fipa-pta)
            list(APPEND _feedingfrenzy2_lto_link -fipa-pta)
        endif()
        foreach(target feedingfrenzy2_recomp feedingfrenzy2)
            if(TARGET ${target})
                target_compile_options(${target} PRIVATE ${_feedingfrenzy2_lto_compile})
            endif()
        endforeach()
        target_link_options(feedingfrenzy2 PRIVATE ${_feedingfrenzy2_lto_link})
        message(STATUS "feedingfrenzy2: Switch LTO on (${FEEDINGFRENZY2_SWITCH_LTO_JOBS} LTRANS jobs, -fipa-pta ${FEEDINGFRENZY2_SWITCH_IPA_PTA})")
    endif()
    if(FEEDINGFRENZY2_SWITCH_LINE_INFO)
        target_compile_options(feedingfrenzy2 PRIVATE -g1 -gdwarf-4)
        target_link_options(feedingfrenzy2 PRIVATE -g1 -gdwarf-4)
    endif()
    # sasrt-nx's CPU round through sa2-nx (2026-10-12). Each has its switch:
    #   - No frame pointer in the game code: AArch64 GCC keeps the x29 frame chain in every non-leaf function by
    #     default, which costs the set-up instruction and one of the callee-saved registers the recompiled code is
    #     short of. The SDK and the app keep theirs (the console profiler's caller chains end at a game function).
    #   - The flush-mode checks the control flow fixes compile to nothing here (tools/flush_mode_flow.py); OFF defines
    #     REX_FLUSH_FLOW_OFF, which puts the original checks back.
    #   - Guest compare-and-swap without the legacy builtin's trailing `dmb ish` (tools/guest_cas.py); OFF defines
    #     REX_GUEST_CAS_LEGACY (the run-time key feedingfrenzy2_cpu_guest_cas_no_barrier puts the barrier back without a build).
    #   - Guest vector loads and stores reversed with REV64 + EXT instead of a table lookup (tools/vec_byteswap.py); OFF
    #     defines REX_BSWAP128_TABLE.
    #   - Constant word shuffles as one permutation (tools/pch_neon_shuffles.py); OFF defines REX_SIMDE_SHUFFLES.
    #   - vperm and the float-to-integer conversions as their NEON instruction (rex/ppc/intrinsics.h); OFF defines
    #     REX_VMX_HELPERS_X86_FORM.
    option(FEEDINGFRENZY2_SWITCH_OMIT_FRAME_POINTER "Game code without the x29 frame chain (Switch)" ON)
    option(FEEDINGFRENZY2_SWITCH_FLUSH_FLOW "Flush-mode checks proven by control flow compile to nothing (Switch)" ON)
    option(FEEDINGFRENZY2_SWITCH_GUEST_CAS "Guest compare-and-swap without the trailing dmb (Switch)" ON)
    option(FEEDINGFRENZY2_SWITCH_VEC_BYTESWAP "Guest vector byte reversal as REV64 + EXT (Switch)" ON)
    option(FEEDINGFRENZY2_SWITCH_NEON_SHUFFLES "Constant word shuffles as one permutation (Switch)" ON)
    option(FEEDINGFRENZY2_SWITCH_NEON_VMX_HELPERS "vperm and float-to-integer conversions as NEON instructions (Switch)" ON)
    set(_feedingfrenzy2_round1_definitions "")
    if(NOT FEEDINGFRENZY2_SWITCH_FLUSH_FLOW)
        list(APPEND _feedingfrenzy2_round1_definitions REX_FLUSH_FLOW_OFF)
    endif()
    if(NOT FEEDINGFRENZY2_SWITCH_GUEST_CAS)
        list(APPEND _feedingfrenzy2_round1_definitions REX_GUEST_CAS_LEGACY)
    endif()
    if(NOT FEEDINGFRENZY2_SWITCH_VEC_BYTESWAP)
        list(APPEND _feedingfrenzy2_round1_definitions REX_BSWAP128_TABLE)
    endif()
    if(NOT FEEDINGFRENZY2_SWITCH_NEON_SHUFFLES)
        list(APPEND _feedingfrenzy2_round1_definitions REX_SIMDE_SHUFFLES)
    endif()
    if(NOT FEEDINGFRENZY2_SWITCH_NEON_VMX_HELPERS)
        list(APPEND _feedingfrenzy2_round1_definitions REX_VMX_HELPERS_X86_FORM)
    endif()
    foreach(target feedingfrenzy2_recomp feedingfrenzy2)
        if(TARGET ${target} AND _feedingfrenzy2_round1_definitions)
            target_compile_definitions(${target} PRIVATE ${_feedingfrenzy2_round1_definitions})
        endif()
    endforeach()
    if(FEEDINGFRENZY2_SWITCH_OMIT_FRAME_POINTER)
        if(TARGET feedingfrenzy2_recomp)
            target_compile_options(feedingfrenzy2_recomp PRIVATE -fomit-frame-pointer)
        endif()
        # The localized copies (app/generated/hot, all compiled in one file of the app) are game code too.
        set_property(SOURCE ${FEEDINGFRENZY2_HOT_SOURCES} APPEND PROPERTY COMPILE_OPTIONS -fomit-frame-pointer)
    endif()
    message(STATUS "feedingfrenzy2: CPU round 1: omit frame pointer ${FEEDINGFRENZY2_SWITCH_OMIT_FRAME_POINTER}, flush flow "
                   "${FEEDINGFRENZY2_SWITCH_FLUSH_FLOW}, guest CAS ${FEEDINGFRENZY2_SWITCH_GUEST_CAS}, vector byteswap "
                   "${FEEDINGFRENZY2_SWITCH_VEC_BYTESWAP}, NEON shuffles ${FEEDINGFRENZY2_SWITCH_NEON_SHUFFLES}, NEON VMX helpers "
                   "${FEEDINGFRENZY2_SWITCH_NEON_VMX_HELPERS}")
endif()

if(REXGLUE_PLATFORM_SWITCH)
    # Profile-guided optimisation (2026-10-14, the user's request for the 1.0.0 release; sa2-nx's setup), for the game
    # code and the app like LTO:
    #   FEEDINGFRENZY2_PGO=generate  an instrumented NRO that writes its branch counts to pgo/ on the SD card
    #                         (src/feedingfrenzy2_pgo.cpp; registered in a section, not by constructors: libgcov's exit-time
    #                         files do not fit the Switch). Value profiling is off: its shared lists are not safe
    #                         without atomic updates.
    #   FEEDINGFRENZY2_PGO=use       the counts merged on the PC (tools/win/pgo_merge.ps1). Code the sessions did not run is
    #                         compiled as without a profile (-fprofile-partial-training), and functions changed since
    #                         the instrumented build keep no profile (-Wno-coverage-mismatch).
    # The profile is pgo/ at the root of the repository (FEEDINGFRENZY2_PGO_DIR): one .gcda per object, named by the object's
    # path in the build folder with # for each folder separator. -fprofile-prefix-path makes GCC use those relative
    # names, so the profile works from any checkout; it must be the build folder written the Windows way, as GCC
    # sees it.
    set(FEEDINGFRENZY2_PGO "" CACHE STRING "Profile-guided optimisation of the Switch build: empty, generate or use")
    set_property(CACHE FEEDINGFRENZY2_PGO PROPERTY STRINGS "" generate use)
    set(FEEDINGFRENZY2_PGO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../pgo" CACHE PATH "Folder of the PGO profile (.gcda files)")
    get_filename_component(_feedingfrenzy2_pgo_dir "${FEEDINGFRENZY2_PGO_DIR}" ABSOLUTE)
    file(TO_NATIVE_PATH "${_feedingfrenzy2_pgo_dir}" _feedingfrenzy2_pgo_dir)
    file(TO_NATIVE_PATH "${CMAKE_BINARY_DIR}" _feedingfrenzy2_pgo_prefix)
    if(FEEDINGFRENZY2_PGO STREQUAL "generate")
        set(_feedingfrenzy2_pgo "-fprofile-generate=${_feedingfrenzy2_pgo_dir}" "-fprofile-prefix-path=${_feedingfrenzy2_pgo_prefix}"
                         -fno-profile-values -fprofile-update=single -fprofile-info-section=gcov_info)
        target_compile_definitions(feedingfrenzy2 PRIVATE FEEDINGFRENZY2_PGO_GENERATE=1)
    elseif(FEEDINGFRENZY2_PGO STREQUAL "use")
        if(NOT EXISTS "${FEEDINGFRENZY2_PGO_DIR}")
            message(FATAL_ERROR "FEEDINGFRENZY2_PGO=use: no profile in ${FEEDINGFRENZY2_PGO_DIR}")
        endif()
        # -fprofile-correction: the counts were taken without atomic updates (-fprofile-update=single) while several
        # game threads ran the same code, so a few increments were lost and some blocks' counts do not add up; GCC
        # smooths them instead of stopping.
        set(_feedingfrenzy2_pgo "-fprofile-use=${_feedingfrenzy2_pgo_dir}" "-fprofile-prefix-path=${_feedingfrenzy2_pgo_prefix}"
                         -fprofile-correction -fno-profile-values -fprofile-partial-training -Wno-missing-profile
                         -Wno-coverage-mismatch)
        # sa2-nx's inliner limits for a build with a profile (measured there: 41 % fewer executed calls for the hot
        # functions 17 % larger; without the unit growth limit the higher one fills the default 40 % with code the
        # session never ran). Empty = the compiler's.
        set(FEEDINGFRENZY2_SWITCH_INLINE_AUTO "64" CACHE STRING
            "max-inline-insns-auto of the game code with a profile (empty: the compiler's)")
        set(FEEDINGFRENZY2_SWITCH_INLINE_UNIT_GROWTH "12" CACHE STRING
            "inline-unit-growth of the game code with a profile (empty: the compiler's)")
        set(_feedingfrenzy2_inline_options "")
        if(FEEDINGFRENZY2_SWITCH_INLINE_AUTO)
            list(APPEND _feedingfrenzy2_inline_options --param=max-inline-insns-auto=${FEEDINGFRENZY2_SWITCH_INLINE_AUTO})
        endif()
        if(FEEDINGFRENZY2_SWITCH_INLINE_UNIT_GROWTH)
            list(APPEND _feedingfrenzy2_inline_options --param=inline-unit-growth=${FEEDINGFRENZY2_SWITCH_INLINE_UNIT_GROWTH})
        endif()
        if(_feedingfrenzy2_inline_options AND TARGET feedingfrenzy2_recomp)
            target_compile_options(feedingfrenzy2_recomp PRIVATE ${_feedingfrenzy2_inline_options})
            set_property(SOURCE ${FEEDINGFRENZY2_HOT_SOURCES} APPEND PROPERTY COMPILE_OPTIONS ${_feedingfrenzy2_inline_options})
        endif()
    elseif(FEEDINGFRENZY2_PGO)
        message(FATAL_ERROR "FEEDINGFRENZY2_PGO must be empty, generate or use (got ${FEEDINGFRENZY2_PGO})")
    endif()
    if(FEEDINGFRENZY2_PGO)
        foreach(target feedingfrenzy2_recomp feedingfrenzy2)
            if(TARGET ${target})
                target_compile_options(${target} PRIVATE ${_feedingfrenzy2_pgo})
            endif()
        endforeach()
        target_link_options(feedingfrenzy2 PRIVATE ${_feedingfrenzy2_pgo})
        message(STATUS "feedingfrenzy2: PGO ${FEEDINGFRENZY2_PGO}")
    endif()

    # The Switch build's fixed settings, compiled in (src/feedingfrenzy2_app.h BuiltInConfig): everything feedingfrenzy2.toml no
    # longer lists. The SD card's feedingfrenzy2.toml holds the player's options and overrides these key by key. Both files
    # are written by tools/make_enhancement_configs.py. (sa2-nx's setup.)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/feedingfrenzy2_switch_defaults.toml" FEEDINGFRENZY2_BUILTIN_CONFIG)

    target_compile_definitions(feedingfrenzy2 PRIVATE FEEDINGFRENZY2_BUILTIN_CONFIG=1)

    # The homebrew menu's entry: name, author, version and icon (app/icon.jpg, 256x256 JPEG, when the project has one).
    set(FEEDINGFRENZY2_VERSION "1.0.0" CACHE STRING "NRO display version" FORCE)
    # feedingfrenzy-nx (2026-10-26): no icon in the repository. Development builds take out/icon_dev.jpg (the game's
